/* =====================================================================================
 * 1. 先看现有项目的分层约定（生成出来的文件必须和现有风格一致）
 *
 *   src/main/java/com/example/demo/
 *     ├── entity/User.java                      -> @TableName + @Data + @TableId
 *     ├── mapper/UserMapper.java                -> extends BaseMapper<User> + @Mapper
 *     ├── service/UserService.java              -> extends IService<User>
 *     ├── service/impl/UserServiceImpl.java     -> extends ServiceImpl<UserMapper, User>
 *     ├── controller/UserController.java        -> @RestController + Result<T> 包装
 *     ├── common/Result.java                    -> 统一返回（已存在，直接用，不要重复生成）
 *     └── config/MybatisPlusConfig.java         -> 分页插件（已存在，不要重复生成）
 *   src/main/resources/
 *     ├── application.yml                       -> 数据源 + mybatis-plus 配置
 *     └── mapper/UserMapper.xml                 -> 只有"多表联查/复杂 SQL"才需要
 *
 *   启动类 DemoSpringbootMysqlApplication 上已有 @MapperScan("com.example.demo.mapper")，
 *   所以新生成的 Mapper 只要放在这个包下，就【不需要】再加 @MapperScan。
 * =====================================================================================
 */

/* =====================================================================================
 * 4. 【方案 A】用 MyBatis-Plus 官方生成器（推荐）
 *
 * 4.1 临时在 pom.xml 的 <dependencies> 中加入（生成完可以删掉，加 test/provided 作用域更干净）：
 *
 *   <dependency>
 *       <groupId>com.baomidou</groupId>
 *       <artifactId>mybatis-plus-generator</artifactId>
 *       <version>3.5.3.1</version>
 *   </dependency>
 *   <dependency>
 *       <groupId>org.apache.velocity</groupId>
 *       <artifactId>velocity-engine-core</artifactId>
 *       <version>2.3</version>
 *   </dependency>
 *
 * 4.2 在 src/test/java/com/example/demo/CodeGenerator.java 写如下生成器入口
 *     （注意：MP 3.5.3 用旧版 FastAutoGenerator 最省事；新版 builder API 字段名不同）：
 *
 *     package com.example.demo;
 *
 *     import com.baomidou.mybatisplus.generator.FastAutoGenerator;
 *     import com.baomidou.mybatisplus.generator.config.OutputConfig;
 *     import com.baomidou.mybatisplus.generator.config.rules.NamingStrategy;
 *     import java.util.Collections;
 *
 *     public class CodeGenerator {
 *         public static void main(String[] args) {
 *             FastAutoGenerator.create(
 *                             "jdbc:mysql://localhost:3306/demo_db?useUnicode=true&characterEncoding=utf8"
 *                                     + "&useSSL=false&serverTimezone=Asia/Shanghai&allowPublicKeyRetrieval=true",
 *                             "root", "1234567890")
 *                     .globalConfig(b -> b
 *                             .author("yourName")
 *                             .disableOpenDir()
 *                             // ★ 输出到项目源码目录（Windows 用反斜杠或双斜杠）
 *                             .outputDir("src/main/java"))
 *                     .packageConfig(b -> b
 *                             .parent("com.example.demo")
 *                             .moduleName("")            // 不要多出 demo 层级
 *                             .entity("entity").mapper("mapper")
 *                             .service("service").serviceImpl("service.impl")
 *                             .controller("controller")
 *                             // ★ XML 不生成到 java 目录，改到 resources/mapper
 *                             .pathInfo(Collections.singletonMap(
 *                                     OutputConfig.xml(), "src/main/resources/mapper")))
 *                     .strategyConfig(b -> b
 *                             .addInclude("t_medical_record")     // ★ 指定表名，可多张
 *                             .addTablePrefix("t_")               // ★ 去掉表前缀
 *                             .entityBuilder()
 *                             .naming(NamingStrategy.underline_to_camel)
 *                             .columnNaming(NamingStrategy.underline_to_camel)
 *                             .logicDeleteColumnName("del_flag")  // 逻辑删除列
 *                             .controllerBuilder().enableRestStyle()
 *                             .enableFileOverride())              // 覆盖旧文件
 *                     .execute();
 *         }
 *     }
 *
 * 4.3 运行：IDEA 里右键 main -> Run（或 mvn -q compile exec:java 之类）。
 *
 * 4.4 ★★ 生成后【必须手工修改】的地方（这是重点，生成器不会替你做）：
 *
 *   (1) 生成器默认输出目录带模块名/绝对路径
 *       -> 确认文件真实落在 src/main/java/com/example/demo/entity 等目录；
 *          若落在项目根的子目录，手工剪切过去。
 *   (2) Lombok
 *       生成器默认产出 getter/setter。项目已用 Lombok，可把实体改成
 *           @Data + @TableName("t_medical_record") + @TableId(value="record_id", type=IdType.AUTO)
 *       （参考 src/main/java/com/example/demo/entity/User.java）。
 *   (3) 主键
 *       自增主键必须显式写 @TableId(type = IdType.AUTO)；
 *       复合主键（如 t_medicine 的 user_id+medicine_id）MP 不支持 @TableId 多主键，
 *       需要自己指定一个业务主键或改用手写 SQL。
 *   (4) 时间/金额类型
 *       把 java.util.Date 改成 LocalDateTime / LocalDate / BigDecimal；
 *       实体里去掉 @JsonFormat 之类生成器附加的注解，保持与 User.java 一致。
 *   (5) Controller 返回体
 *       生成器返回裸对象/AjaxResult，项目统一用 com.example.demo.common.Result<T>：
 *           return Result.ok(service.xxx());   // 成功
 *           return Result.fail("xxx");         // 失败
 *       并删掉生成器自带的 @RequestMapping 前缀冲突（参考 UserController 的写法）。
 *   (6) Mapper 注解
 *       项目靠启动类 @MapperScan 扫描，生成器多加的 @Mapper / @Repository 可保留也可删，
 *       不要重复添加多个扫描方式。
 *   (7) XML
 *       生成器会生成一份空 <mapper> XML。application.yml 的
 *       mapper-locations=classpath*:/mapper/**/*.xml 已能扫到；
 *       若没有自定义 SQL，可以直接删除该 XML（MP 单表 CRUD 不需要 XML）。
 *   (8) Service 接口包路径
 *       生成器可能生成 service/impl 到 com/example/demo/service/impl，确认与现有
 *       UserServiceImpl 同级；接口上不要继承生成器自定义的 IService 变体。
 *   (9) 依赖清理
 *       生成完成后把 mybatis-plus-generator / velocity-engine-core 从 pom 移除或改 scope，
 *       避免打进生产 jar。
 *  10  编译验证：mvn -q -DskipTests compile 通过后再启动。
 * =====================================================================================
 */

/* =====================================================================================
 * 5. 【方案 B】手工模板（一）—— Entity
 *    文件：src/main/java/com/example/demo/entity/MedicalRecord.java
 * =====================================================================================
package com.example.demo.entity;

import com.baomidou.mybatisplus.annotation.IdType;
import com.baomidou.mybatisplus.annotation.TableId;
import com.baomidou.mybatisplus.annotation.TableLogic;
import com.baomidou.mybatisplus.annotation.TableName;
import lombok.Data;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;

@Data
@TableName("t_medical_record")
public class MedicalRecord {

    @TableId(value = "record_id", type = IdType.AUTO)
    private Long recordId;              // record_id  主键

    private String patientName;         // patient_name 患者姓名

    private Integer patientAge;         // patient_age 患者年龄

    private LocalDate visitDate;        // visit_date   就诊日期

    private String diagnosis;           // diagnosis    诊断结论

    private BigDecimal totalFee;        // total_fee    费用

    @TableLogic
    private Integer delFlag;            // del_flag     0-正常 1-已删除

    private String createUser;          // create_user  创建人

    private LocalDateTime createTime;   // create_time  创建时间

    private String updateUser;          // update_user  更新人

    private LocalDateTime updateTime;   // update_time  更新时间
}
 *
 * ★ 需要改的地方：
 *   - @TableName 必须写真实表名（表名与类名不完全等价时必须写）。
 *   - 主键列名不是 id 时，@TableId 要写 value。
 *   - 有 del_flag 才加 @TableLogic；同时在 application.yml 里配置
 *         mybatis-plus.global-config.db-config.logic-delete-field: delFlag
 *         mybatis-plus.global-config.db-config.logic-delete-value: 1
 *         mybatis-plus.global-config.db-config.logic-not-delete-value: 0
 *     否则 @TableLogic 不生效（本项目当前只配了 id-type: auto）。
 *   - 字段名不符合"下划线转驼峰"时补 @TableField("真实列名")。
 * =====================================================================================
 */

/* =====================================================================================
 * 7. 【方案 B】手工模板（三）—— Mapper XML（无自定义 SQL 时可跳过）
 *    文件：src/main/resources/mapper/MedicalRecordMapper.xml
 * =====================================================================================
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.example.demo.mapper.MedicalRecordMapper">

    <select id="selectByPatientName" resultType="com.example.demo.entity.MedicalRecord">
        SELECT record_id, patient_name, patient_age, visit_date, diagnosis, total_fee
        FROM t_medical_record
        WHERE del_flag = 0
        <if test="patientName != null and patientName != ''">
            AND patient_name LIKE CONCAT('%', #{patientName}, '%')
        </if>
        ORDER BY create_time DESC
    </select>

</mapper>
 *
 * ★ 需要改的地方：
 *   - namespace 必须等于 Mapper 接口全限定名，否则会 "Invalid bound statement"。
 *   - <select id> 必须等于接口方法名。
 *   - 返回多表结果时 resultType 指向 dto 包（application.yml 的
 *     type-aliases-package: com.example.demo.dto 已配置别名）。
 *   - 路径必须在 resources/mapper/ 下，与 mapper-locations 匹配。

/* =====================================================================================
 * 9. 【方案 B】手工模板（五）—— Controller

package com.example.demo.controller;

import com.baomidou.mybatisplus.core.metadata.IPage;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.example.demo.common.Result;
import com.example.demo.entity.MedicalRecord;
import com.example.demo.service.MedicalRecordService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/medicalRecord")
public class MedicalRecordController {

    @Autowired
    private MedicalRecordService medicalRecordService;

    @GetMapping("/page")
    public Result<IPage<MedicalRecord>> page(@RequestParam(defaultValue = "1") long current,
                                             @RequestParam(defaultValue = "10") long size) {
        return Result.ok(medicalRecordService.page(new Page<>(current, size)));
    }

    @GetMapping("/{id}")
    public Result<MedicalRecord> getById(@PathVariable Long id) {
        return Result.ok(medicalRecordService.getById(id));
    }

    @PostMapping("/save")
    public Result<Boolean> save(@RequestBody MedicalRecord record) {
        return Result.ok(medicalRecordService.save(record));
    }

    @PutMapping("/update")
    public Result<Boolean> update(@RequestBody MedicalRecord record) {
        return Result.ok(medicalRecordService.updateById(record));
    }

    @DeleteMapping("/{id}")
    public Result<Boolean> remove(@PathVariable Long id) {
        return Result.ok(medicalRecordService.removeById(id));
    }
}
 *
 * ★ 需要改的地方：
 *   - 返回值统一包 com.example.demo.common.Result，不要返回裸对象。
 *   - 分页必须用 MP 的 Page/IPage（分页插件已在 config/MybatisPlusConfig 注册，无需再加）。
 *   - 路径前缀 /medicalRecord 与前端约定保持一致，避免与已有 Controller 冲突。
 * =====================================================================================
 */

/* =====================================================================================
 * 10. 生成完之后的"必查清单"（按顺序过一遍）
 *   [ ] pom.xml          是否临时加了 generator/velocity，用完是否移除

 *   [ ] XML              namespace、id、resultType；无自定义 SQL 可删

 *   [ ] Controller       Result 包装、请求路径、分页对象
